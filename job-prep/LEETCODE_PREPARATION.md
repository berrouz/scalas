## 📝 Scala-Specific Tips for LeetCode

### Use Functional Style
```scala
// Instead of imperative loops, use:
nums.foldLeft(0)(_ + _)
nums.sliding(2).map(_.sum)
nums.zipWithIndex.filter(_._1 > 0)

// Pattern matching for trees
def maxDepth(root: TreeNode): Int = root match {
  case null => 0
  case node => 1 + math.max(maxDepth(node.left), maxDepth(node.right))
}

// Recursion with tail optimization
@tailrec
def binarySearch(arr: Array[Int], target: Int, lo: Int, hi: Int): Int = {
  if (lo > hi) -1
  else {
    val mid = lo + (hi - lo) / 2
    if (arr(mid) == target) mid
    else if (arr(mid) < target) binarySearch(arr, target, mid + 1, hi)
    else binarySearch(arr, target, lo, mid - 1)
  }
}
```

### Common Scala Patterns
```scala
// Two Sum with foldLeft
def twoSum(nums: Array[Int], target: Int): Array[Int] = {
  nums.zipWithIndex.foldLeft(Map.empty[Int, Int]) { case (map, (num, i)) =>
    map.get(target - num) match {
      case Some(j) => return Array(j, i)
      case None => map + (num -> i)
    }
  }
  Array.empty
}

// Group Anagrams with groupBy
def groupAnagrams(strs: Array[String]): List[List[String]] = {
  strs.groupBy(_.sorted).values.map(_.toList).toList
}

// BFS with Queue
import scala.collection.mutable.Queue
def bfs(root: TreeNode): List[Int] = {
  val queue = Queue(root)
  val result = List.newBuilder[Int]
  while (queue.nonEmpty) {
    val node = queue.dequeue()
    if (node != null) {
      result += node.value
      queue.enqueue(node.left, node.right)
    }
  }
  result.result()
}
```
